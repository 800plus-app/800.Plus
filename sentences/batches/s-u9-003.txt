# יחידה 9 · פריטים 121-180

121	שׁוּמָה	אומדן ערך, הערכה; גידול שפיר על העור
122	תִּמּוּכִין	סימוכין וגיבוי לטענה
123	תַּעַר	סכין גילוח; נדן החרב
124	סְפּוֹרָדִי	אקראי ולא סדיר
125	לָקוֹנִי	קצר ויבש בניסוחו
126	קוֹהֶרֶנְטִי	עקבי ומגובש, נטול סתירות
127	לָטֶנְטִי	סמוי ורדום
128	אֶמְפִּירִי	מבוסס ניסיון ותצפית
129	אַסְקֻפָּה נִדְרֶסֶת	אדם מושפל שכולם מתעמרים בו
130	עֵינַיִם טְרוּטוֹת	עיניים עייפות ואדומות
131	חֲדַל אִישִׁים	אדם בזוי וחסר ערך
132	חָזוּת קָשָׁה	נבואה קודרת על הבאות
133	חָזוֹן נִפְרָץ	תופעה רווחת ושגרתית
134	כִּצְנִינִים בְּעֵינָיו	מטרד ומקור סבל עבורו
135	שִׁבְעָה מוֹנִים	הרבה יותר, במידה מוגברת
136	אַחַת דָּתוֹ	דינו נחרץ ואין מנוס
137	בְּהֵן צֶדֶק	בהבטחת אמת מחייבת, במילת כבוד
138	אַכְזָב	שאינו מתמיד (נחל אכזב)
139	חֲלוֹם בְּאַסְפַּמְיָא	משאלה רחוקה מהמציאות
140	בַּלָּאט	בחשאי, בסתר
141	חָרִיט	ארנק, כיס קטן לכסף
142	עִנְבָּל	לשון הפעמון
143	קוּף הַמַּחַט	הנקב להשחלת החוט
144	שַׁבְשֶׁבֶת	מכשיר המראה את כיוון הרוח
145	עֱלִי	מוט הכתישה במדוכה
146	קַת	ידית הנשק או הכלי
147	טֶפַח	מידת רוחב כף יד (מגלה טפח)
148	דַּיָּה	עוף דורס בינוני ממשפחת הנצים
149	זַרְזִיר	ציפור נודדת בלהקות
150	עָלָה בְּקָנֶה אֶחָד	תאם, לא סתר
151	גִּילְדָה	אגודת בעלי מקצוע
152	מִנְשָׁר	מניפסט, הצהרת עקרונות פומבית
153	גְּלִימָה	מעיל עליון ארוך
154	בֵּית נְכוֹת	מוזיאון
155	מוֹךְ	חומר ריפוד רך
156	דַּפְתָּר	פנקס רישום
157	אִמְרָה	פתגם; שולי הבגד
158	זוּלָתָן	מעדיף את טובת האחר
159	סַנְוֵרִים	עיוורון פתאומי מאור חזק (הוכה בסנוורים)
160	עֲוִית	התכווצות שרירים פתאומית
161	חֲכִירָה	שכירת קרקע לתקופה קצובה
162	קוֹלָן	מכשיר לכיוון כלי נגינה
163	מֶלְקָחַיִם	כלי אחיזה וצביטה
164	רַכָּב	נוהג הכרכרה
165	תִּיתוֹרָה	שולי הכובע הבולטים
166	כֵּלִים שְׁלוּבִים	מערכות המשפיעות זו על זו
167	יָדוֹ עַל הַתַּחְתּוֹנָה	הוא המפסיד במאבק
168	מֵאֵין כָּמוֹהוּ	שאין דומה לו
169	סִדְקִית	חפצי תפירה קטנים
170	עַל קְצֵה הַמַּזְלֵג	בתמצית בלבד
171	הֶחֱזִיר עֲטָרָה לְיָשְׁנָהּ	השיב דבר לתפארתו הקודמת
172	אֵין עָלָיו עוֹרְרִין	אין חולק על כך
173	סְיָח	סוס צעיר
174	אוֹתִיּוֹת קִדּוּשׁ לְבָנָה	כתב גדול ובולט במיוחד
175	מַרְבֵּק	תא פיטום לבהמות (עגל מרבק)
176	הִתְעַתֵּד	עמד לעשות דבר, נערך לקראתו
177	בֶּן דְּמוּתוֹ	הדומה לו כשתי טיפות
178	לַבְלָר	כתבן ופקיד רישום
179	שָׂרִיג	ענף גפן
180	סְמָדַר	פריחת הגפן, ענבים בתחילת גידולם
