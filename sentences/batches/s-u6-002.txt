88	בֶּן לְוָיָה	שותף לדרך
89	בַּת קוֹל	הד; הכרזה משמיים

91	בָּבַת עַיִן	אישון העין; דבר יקר מכול (שמר עליו כבבת עינו)
92	בֶּן טִפּוּחָיו	חניך אהוב שגידלו אותו במסירות
93	כְּלִיל הַשְּׁלֵמוּת	נטול פגמים, מצוין לחלוטין
94	הִדְרִיךְ מְנוּחָתוֹ	הטריד ולא הרפה ממנו
95	פָּרַק עֹל	השתחרר מכל מרות
96	טָמַן לוֹ פַּח	הכין לו מלכודת
97	הִרְעִישׁ עוֹלָמוֹת	עורר מהומה גדולה
98	פַּח יָקוּשׁ	מלכודת נסתרת
99	דֶּרֶךְ הַמֶּלֶךְ	הדרך הראשית והמקובלת
100	בָּטֵל בְּשִׁשִּׁים	זעיר עד חוסר חשיבות
101	יְקַר הַמְּצִיאוּת	נדיר ביותר
102	פִּיק בִּרְכַּיִם	רעד ברכיים מאימה
103	שְׁתִי וָעֵרֶב	לכל הכיוונים (חצה את העיר שתי וערב)
104	נִכְנַס בְּעֹבִי הַקּוֹרָה	צלל לעומק העניין
105	אֶרֶץ גְּזֵרָה	מקום נידח של גלות
106	עַז נֶפֶשׁ	אמיץ, חסר מורא
107	עַזּוּת מֵצַח	חוצפה גמורה
108	גַּס רוּחַ	חסר נימוס ועדינות
109	דֶּרֶךְ אֶרֶץ	נימוס והתחשבות
110	שְׁוֵה נֶפֶשׁ	אדיש, לא מתרגש
111	מִדָּה כְּנֶגֶד מִדָּה	גמול התואם את המעשה
112	כְּיַד הַמֶּלֶךְ	בשפע וברוחב לב
113	קוֹרַת גַּג	בית ומחסה
114	אֶבֶן שׁוֹאֶבֶת	מוקד המושך רבים
115	גֻּלַּת הַכּוֹתֶרֶת	החלק המרשים ביותר
116	בְּגַפּוֹ	לבדו, בלי בני לוויה
117	בְּמַפְגִּיעַ	בתוקף, בהדגשה רבה
118	בַּעֲלִיל	באופן ברור, בלי ספק
119	בּוֹשׁ	נכלם, חש מבוכה
