# יחידה 8 · פריטים 121-180

121	קָשַׁר לוֹ כְּתָרִים	הרעיף עליו שבחים
122	בֵּית מַרְזֵחַ	מסבאה, פונדק שתייה
123	הִקְדִּיחַ	חרך, שרף מחום רב (הקדיח את תבשילו)
124	עָגְמַת נֶפֶשׁ	צער, עצב ודכדוך
125	סְגֻלָּה	מעלה מיוחדת; תרופה עממית
126	הִלָּה	עטרת זוהר; יוקרה סביב אדם
127	תַּהְפּוּכוֹת	שינויי גורל חדים
128	מִקְלַעַת	צמה, מעשה שזירה
129	נִדָּח	מרוחק ושכוח (כפר נידח)
130	הִתְנוֹסֵס	התנופף בגאון (הדגל התנוסס)
131	עַד מְהֵרָה	תוך זמן קצר
132	עַל נְקַלָּה	בלי מאמץ, בלי קושי
133	דִּבָּה	השמצה ולעז (הוציא דיבה)
134	פָּרִיךְ	שביר ומתפורר בקלות
135	עֲרוּגָה	שטח שתילה תחום בגן
136	אֶבְרָה	כנף גדולה של עוף
137	נְכָלִים	מזימות ותככים
138	אָבָה	הסכים ורצה (לא אבה לשמוע)
139	הוֹאִיל	ניאות ברצונו הטוב (הואיל בטובו)
140	נָקְעָה נַפְשׁוֹ	מאס לחלוטין
141	רָחַשׁ כָּבוֹד	הרגיש הערכה עמוקה
142	נָצַר לְשׁוֹנוֹ	נזהר בדיבורו ושתק
143	קָצְרָה יָדוֹ	לא היה ביכולתו
144	נָפַל דָּבָר	אירע מאורע מכריע
145	פְּרִי הִלּוּלִים	תוצר מעולה שראוי להערכה
146	הַשְׁכֵּם וְהַעֲרֵב	ללא הרף, בכל שעה
147	בֵּין הָעַרְבַּיִם	שעת דמדומים לפני רדת החשכה
148	אִישׁוֹן לַיְלָה	שעה מאוחרת בעומק החשכה
149	מְאוֹר פָּנִים	חביבות ולבביות כלפי הזולת
150	בְּרֹב עָם	בנוכחות קהל גדול
151	גַּרְדּוֹם	במת הוצאה להורג
152	בַּרְדָּס	כיסוי ראש מחובר לבגד
153	מַחְתָּה	כלי לגריפת גחלים
154	דְּחִי	כישלון ומעידה (מדחי אל דחי)
155	פָּעָה	השמיע קול כבש
156	חֶלְכָּאִי	עני עלוב ונדכא
157	פִּיד	שבר, חורבן פתאומי
158	סַד	מתקן קיבוע וכליאה
159	רָשׁ	עני מרוד (כבשת הרש)
160	שַׁחַל	כינוי מקראי לאריה טורף
161	יַעֲרָה	חלת דבש
162	לַיִשׁ	אריה זקן ואיתן
163	טֶלֶף	פרסת בעל חיים
164	יָצוּל	מוט בעגלה שאליו רותמים בהמה
165	נְחֻשְׁתַּיִם	כבלי אסירים
166	פַּדַּחַת	חלק הגולגולת שמעל המצח
167	שָׁבִיס	כיסוי ראש לנשים
168	קְנוֹקֶנֶת	גבעול מסולסל שבו נאחז צמח מטפס
169	צַלְיָן	עולה לרגל למקום קדוש
170	קֶבֶס	בחילה וגועל
171	קַרְדֹּם	גרזן לחטיבת עצים
172	זֶרֶד	ענף דק
173	אָמִיר	צמרת העץ
174	פֹּארָה	ענף עץ או גפן
175	דְּבֵלָה	גוש תאנים מיובשות
176	מִבְדּוֹק	מתקן תיקון ספינות
177	מַחְבֵּצָה	כלי להכנת חמאה
178	שְׁדֵמָה	שדה תבואה
179	מִכְמֹרֶת	רשת דיג גדולה
180	קֶסֶת	כלי קיבול לדיו
