# יחידה 4 · פריטים 1-60

1	הֵמִיט	הביא רעה על מישהו (המיט אסון)
2	אֲמַתְלָה	תירוץ מזויף
3	שָׁצַף	זרם בעוצמה, גאה (בשצף קצף)
4	הִדִּיר רַגְלָיו	נמנע מלבוא למקום, הרחיק את עצמו ממנו
5	כֵּרָה	סעודה, משתה
6	בָּטַשׁ	בעט, רמס, רקע ברגליו
7	כִּלָּה	השמיד לגמרי (כילה את זעמו)
8	חָצַב	כרת מן הסלע
9	סִתֵּת	עיבד וחתך אבן
10	גִּלֵּף	חתך צורות ועיטורים בעץ
11	חָרַט	חקק בחומר קשה
12	יָצַק	שפך נוזל או מתכת; מילא (יצק תוכן)
13	הִתִּיךְ	המיס בחום גבוה
14	מֵרֵק	צחצח והבריק
15	אָרַג	יצר בד מחוטים
16	פָּסַח	דילג מעל (לא פסח על איש)
17	קִפֵּחַ	מנע את המגיע לו, הפלה לרעה; איבד (קיפח את חייו)
18	רָדָה	שלט ביד קשה (רדה בו); הוציא דבש (רדה דבש)
19	מָחָה	הביע התנגדות; ניגב (מחה דמעה)
20	הִלְעִיט	האכיל בכמות מופרזת (הלעיט במידע)
21	סָבָא	שתה לשוכרה (זולל וסובא)
22	הִתְבַּסֵּם	השתכר קלות (התבסם מהיין)
23	נָסַךְ	החדיר תחושה (נסך ביטחון); יצק יין בפולחן
24	תָּסַס	בעבע ופלט גזים; געש מהתרגשות (נעורים תוססים)
25	הִבְאִישׁ	הסריח; השמיץ (הבאיש את ריחו)
26	נָדַף	הפיץ ריח (נדף ניחוח); נישא ברוח
27	הִתִּיז	פיזר רסיסים; כרת בתנופה (התיז את ראשו)
28	נִקְוָה	התאסף במקום (מים נקוו)
29	קָד	השתחווה קלות (קד קידה)
30	סָגַד	כרע והשתחווה (סגד לפסל); העריץ עד התבטלות
31	זִקֵּק	טיהר והפריד בהרתחה (זיקק נפט)
32	דִּשְׁדֵּשׁ	הלך בכבדות; לא התקדם (דשדש במקום)
33	שִׁרְבֵּב	הבליט החוצה (שרבב לשונו); הכניס בטעות (שורבב שיבוש)
34	כִּרְכֵּר	קיפץ, רקד והסתובב (כרכר סביבו)
35	שָׁעַט	דהר ברעש (שעטת סוסים)
36	נִתַּר	קפץ בפתאומיות, זינק (ניתר ממקומו)
37	חִשֵּׁל	חיזק והקשיח (חושל בקרבות); עיבד מתכת בהכאה
38	הִתְרַפֵּק	נצמד בחיבה; נזכר בגעגוע (התרפק על העבר)
39	חָשַׁק	השתוקק (חשקה נפשו); הידק (חשק שיניים)
40	נִחַם	התחרט (ניחם על מעשהו)
41	הִטִּיחַ	זרק בכוח (הטיח בקיר); אמר בבוטות (הטיח האשמות)
42	מֵרֵר	הרע והציק (מירר את חייו); בכה קשות (מירר בבכי)
43	הִתְרַצָּה	נעתר והסכים לבסוף
44	הִתְבַּטֵּל	התעצל ובזבז זמן; החשיב עצמו כאין (התבטל בפניו)
45	רִתֵּק	קשר למקום (רותק למיטה); משך תשומת לב רבה
46	שָׁבָה	לכד אדם והחזיק בו בכוח (שבה חיילים)
47	הִתְפַּרְקֵד	שכב פרוש על גבו
48	שָׁכַךְ	נחלש ונרגע (הכעס שכך)
49	גָּוַע	מת (גווע ברעב)
50	גָּסַס	נטה למות
51	נָבַר	חיטט וחיפש (נבר במגירות)
52	פִּשְׁפֵּשׁ	חיפש ביסודיות, בדק ובחן (פשפש במעשיו)
53	נֶחְלַץ	יצא מסכנה או ממצוקה; נרתם לעזרה (נחלץ לעזרתו)
54	כִּלְכֵּל	פרנס וקיים; תכנן (כלכל את צעדיו)
55	פָּרַע	שילם חוב (פרע הלוואה); הפר סדר (פרע פרעות)
56	נָשָׁה	תבע חוב (נושה בחברו)
57	חָב	היה עליו לשלם או לגמול (חב לו תודה)
58	גָּרַף	אסף בכלי (גרף עלים); צבר בשפע (גרף רווחים)
59	הֶחֱטִיא	פספס את המטרה; הכשיל (החטיא את הרבים)
60	הָדַף	דחף לאחור (הדף את המתקפה)
