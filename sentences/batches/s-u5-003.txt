# יחידה 5 · פריטים 121-180

121	נֶאֱלָח	מתועב ומושחת
122	שָׁחוּחַ	כפוף, רכון, מורכן (גב שחוח)
123	מְאַלֵּף	מחכים ומלמד (דוגמה מאלפת)
124	הֵנִיחַ אֶת דַּעְתּוֹ	הרגיע וסיפק אותו
125	הוֹלִיךְ שׁוֹלָל	רימה והטעה
126	הֵקֵל רֹאשׁ	זלזל, לא התייחס ברצינות
127	מִן הַפַּח אֶל הַפַּחַת	ממצב רע לגרוע ממנו
128	אֶפֶס קָצֵהוּ	רק מעט מזעיר ממנו
129	טִפִּין טִפִּין	מעט מעט, בהדרגה
130	אֵין אוֹנִים	חסר כוח וישע
131	תְּמוֹל שִׁלְשׁוֹם	בעבר הקרוב (לא כתמול שלשום)
132	עוּל יָמִים	צעיר מאוד
133	שְׁאֵר בָּשָׂר	קרוב משפחה
134	רְחַב יָד	נדיב מאוד
135	אֶרֶךְ אַפַּיִם	סבלני, מאופק בכעסו
136	חֲמוּר סֵבֶר	רציני וקפדני בהבעתו
137	נִים וְלֹא נִים	מנומנם למחצה
138	עַתִּיק יוֹמִין	קדום מאוד
139	שֹׁד וָשֶׁבֶר	קריאת אסון וזעזוע
140	מִן הַכֹּחַ אֶל הַפֹּעַל	ממחשבה למעשה
141	מְחוֹז חֶפְצוֹ	היעד הנכסף
142	בִּמְרוּצַת הַזְּמַן	עם חלוף הזמן
143	פָּרָשַׁת דְּרָכִים	צומת; נקודת הכרעה
144	רֹעַ הַגְּזֵרָה	הפור הרע שנפל (השלים עם רוע הגזרה)
145	עַמּוּד הַתָּוֶךְ	המרכזי והחשוב ביותר
146	אֶבֶן בֹּחַן	קנה מידה מכריע להערכה
147	מַטְבֵּעַ לָשׁוֹן	ביטוי קבוע בשפה
148	בַּעַל שִׁעוּר קוֹמָה	אדם רם מעלה ומוערך
149	נְשׂוּא פָּנִים	מכובד ורם מעלה
150	רַב פְּעָלִים	עתיר מעשים, חרוץ ונמרץ
151	קִבֵּל עָלָיו אֶת הַדִּין	השלים עם הגזרה
152	דָּבָר וְהִפּוּכוֹ	צמד ניגודים מוחלטים
153	לֹא דֻּבִּים וְלֹא יַעַר	דבר שלא היה כלל
154	דָּן לְכַף זְכוּת	שפט לטובה
155	לִפְנִים מִשּׁוּרַת הַדִּין	במידת חסד, מעבר לחובה
156	יָצָא יְדֵי חוֹבָה	עשה רק את הנדרש
157	עוֹבֵר לַסּוֹחֵר	מקובל על הכול (מטבע עובר לסוחר)
158	שָׁוֶה לְכָל נֶפֶשׁ	פשוט ונגיש לכל אדם
159	לֶחֶם חֻקּוֹ	מזון יומי ואמצעי מחיה; דבר שבשגרה אצלו
160	קֻצּוֹ שֶׁל יוֹד	פרט קטן שבקטנים
161	מִלֵּא פִּיו מַיִם	שתק ולא הגיב
162	לְשׁוֹן הַמֹּאזְנַיִם	הגורם המכריע בין שקולים
163	קֶרֶן זָוִית	פינה נסתרת (נדחק לקרן זווית)
164	עָמַד בַּפֶּרֶץ	התייצב למנוע אסון
165	אַחַר כָּבוֹד	כביכול בכבוד (סולק אחר כבוד)
166	בַּחֲצִי פֶּה	בהיסוס ובלא רצון
167	בְּיָד רָמָה	בגלוי ובביטחון מופגן
168	בְּאֵין רוֹאִים	בחשאי, בלי שיבחינו בו
169	בֵּין הַשְּׁמָשׁוֹת	שעת הדמדומים
170	לְעֵת מְצֹא	כשמזדמן, בשעת הצורך
171	נַחֲלַת הַכְּלָל	רכוש ציבורי, נכס פתוח לשימוש הרבים
172	נִכְסֵי צֹאן בַּרְזֶל	ערכי יסוד שאין לוותר עליהם
173	יְגִיעַ כַּפָּיו	עבודתו הקשה; פרי עמלו
174	הֲלָךְ רוּחַ	מצב נפשי, אווירה
175	עִדָּן וְעִדָּנִים	זמן רב מאוד
176	עָשָׂה יָמִים כְּלֵילוֹת	עבד בלי הפסקה
177	עֵרֶב רַב	אספסוף, קהל מגוון וחסר ייחוד
178	לְמַגִּנַּת לִבּוֹ	לצערו העמוק
179	אַלְיָה וְקוֹץ בָּהּ	טובה שיש בה פגם
180	לַאֲמִתּוֹ שֶׁל דָּבָר	למעשה, בעצם
