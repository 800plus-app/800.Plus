# יחידה 4 · פריטים 181-208

181	שְׁכוּחַ אֵל	נידח ועזוב
182	עוֹלֶה עַל גְּדוֹתָיו	מלא עד בלי הכיל
183	צַר עַיִן	מקנא בטובת זולתו, קמצן
184	בַּר חֲלוֹף	זמני, שסופו לעבור
185	נוֹשֵׂא כֵּלִים	עוזר צמוד ונאמן
186	אָבַד עָלָיו הַכֶּלַח	התיישן לגמרי
187	יָצָא בְּשֵׁן וָעַיִן	ניצל בקושי ובהפסד כבד
188	כְּמוֹצֵא שָׁלָל רָב	בשמחה עצומה על דבר יקר
189	הִגְדִּישׁ אֶת הַסְּאָה	הגזים מעבר לנסבל
190	יָצָא מִגִּדְרוֹ	התרגש עד אובדן שליטה; עשה מאמץ מיוחד
191	נֶחְבָּא אֶל הַכֵּלִים	צנוע ומצטנע
192	הִתְרָה	הזהיר
193	הִדְבִּיר	הכניע, חיסל; השמיד מזיקים
194	כֶּבֶשׁ	מישור משופע לעלייה וירידה, רמפה
195	מֵצֵר	מצטער על משהו
196	בָּזַק	פיזר, הפיץ, זרה
197	לְמוֹתָר	אין בו צורך, לא נחוץ
198	בָּא עַל שְׂכָרוֹ	קיבל את הגמול המגיע לו, זכה לתמורה
199	אֵיתָן בְּדַעְתוֹ	תקיף, החלטי, אינו נסוג מעמדתו
200	סִיקֵּל	פינה אבנים, ניקה את השטח מאבנים
201	יַנְקוּת	ראשית החיים, הגיל הרך
202	מְגֻלָּם	טמון, כלול; מוצג בידי שחקן
203	לֹא יָדַע אֶת נַפְשׁוֹ	היה נסער עד כדי בלבול
204	רָחַץ בְּנִקְיוֹן כַּפָּיו	טען לחפותו
205	שָׂם נַפְשׁוֹ בְּכַפּוֹ	הסתכן מאוד
206	פָּשַׁט יָדוֹ	ביקש צדקה מעוברים ושבים
207	תְּהֵא אֲשֶׁר תְּהֵא	לא משנה איזו
208	הִקְשָׁה עָרְפּוֹ	התעקש, סירב לשמוע
