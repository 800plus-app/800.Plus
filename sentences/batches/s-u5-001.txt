# יחידה 5 · פריטים 1-60

1	חִישׁ	במהירות (חיש קל)
2	אַט	בנחת, במתינות (אט אט)
3	דּוּמָם	בשקט, בלי תנועה (עמד דומם)
4	בְּפַרְהֶסְיָה	בפומבי, לעיני כול
5	בְּצִנְעָה	בסתר, הרחק מעיני הציבור (נערך בצנעה)
6	בַּחֲשַׁאי	בסודיות (פעל בחשאי)
7	בְּעִתּוֹת	בזמנים, בתקופות (בעתות משבר)
8	קְבַרְנִיט	רב חובל; מנהיג (קברניטי המשק)
9	נוֹגֵשׂ	מעביד קשה ואכזר
10	יָשִׁישׁ	זקן מופלג
11	טַף	ילדים רכים (נשים וטף)
12	זַאֲטוּט	ילד קטן
13	קֻנְדָּס	שובב ותעלולן (מעשי קונדס)
14	אֶפִּיקוֹרוֹס	כופר ומטיל ספק
15	צָלַח	חצה, עבר מכשול (צלח את הנהר); עלה יפה
16	מִלֵּל	אמר, דיבר (מי פילל ומי מילל)
17	עִלְעֵל	דפדף ברפרוף (עלעל בספר)
18	נָשַׁף	פלט אוויר מהפה (נשף בעורפו)
19	אָרָה	קטף פירות (ארה תאנים)
20	תִּעְתֵּעַ	הטעה במכוון (תעתע ביריבו)
21	טִשְׁטֵשׁ	עמעם והסתיר (טשטש ראיות)
22	לִבֵּן	בירר עד תום (ליבנו את המחלוקת)
23	לִפֵּף	כרך וגלגל סביב (ליפף תחבושת)
24	גָּלַל	הניע חפץ בסיבוב; סיפר בפירוט (גולל פרשה)
25	מִשֵּׁשׁ	בדק בידיים (מישש את הסחורה)
26	גִּשֵּׁשׁ	חיפש בזהירות (גישש את דרכו)
27	רִקֵּעַ	עיבד מתכת בהקשה (רוקע זהב)
28	שָׁעָה	הקשיב ונענה (לא שעה לאזהרות)
29	שִׁפֵּד	נעץ ותקע (שיפד את יריבו בדבריו)
30	הֶעְפִּיל	טיפס במאמץ; הגיע לשלב גבוה (העפיל לגמר)
31	הֻגְלָה	נשלח בכפייה מארצו
32	עָקַד	קשר איברים (עקדת יצחק)
33	הִסָּה	השתיק (היסה את ההמון)
34	הִקְשָׁה	הכביד וסיבך; שאל בהתנגדות (הקשה על דבריו)
35	זִלֵּף	התיז טיפות (זילף בושם)
36	גָּחַן	רכן והתכופף קדימה (גחן אל התינוק)
37	בִּתֵּר	חתך לחלקים (ביתר את השטח)
38	פִּכָּה	נבע וזרם החוצה (מעיין מפכה)
39	קָרַשׁ	התקשה מנוזל (דם קרוש)
40	הִשִּׂיא	חיתן, קידש בני זוג (השיא את בתו)
41	נוֹעַץ	ביקש חוות דעת (נועץ במומחה)
42	הִתְגּוֹדֵד	התקהל בצפיפות (התגודדו סביבו)
43	הִתְעָרָה	נקלט והשתרש (התערה בחברה)
44	נִטְמַע	נבלע והתמזג (נטמע בסביבה)
45	סָר	פנה ועזב (סר מהדרך); ביקר (סר לביתו)
46	הִפְשִׁיל	קיפל כלפי מעלה (הפשיל שרוולים)
47	אָזַר	אסף כוח (אזר עוז)
48	חָגַר	קשר למותניו (אל יתהלל חוגר כמפתח)
49	הִתְנַגֵּחַ	התעמת בחריפות
50	סִלֵּף	עיוות במתכוון (סילף את דבריו)
51	גָּעָה	השמיע קול בהמה; פרץ בבכי (געה בבכייה)
52	צָהַל	שמח בקול; צנף כסוס (הסוס צהל)
53	שָׁרַץ	מלא ומתרבה (המקום שורץ)
54	תָּפַח	התנפח וגדל (הבצק תפח)
55	פִּרְכֵּס	נע בעוויתות; התקשט (פרכסה עצמה)
56	שִׁלְשֵׁל	הוריד מטה (שלשל מטבע לקופה)
57	נִפֵּץ	שבר לרסיסים (ניפץ אשליה)
58	הִתְפָּעֵם	התרגש והתמלא רגש (נפעם)
59	הִבְלִיג	כבש רגשותיו ולא הגיב
60	הִפְרִישׁ	הבדיל והקצה (הפריש תרומה)
