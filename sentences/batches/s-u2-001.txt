# יחידה 2 · פריטים 1-60

1	בְּעֶטְיוֹ	בגללו, באשמתו
2	פֶּן	שמא, כדי שלא
3	לְבַל	כדי שלא
4	בְּטֶרֶם	לפני ש־
5	בִּן־לַיְלָה	בבת אחת, במהירות
6	זֶה מִכְּבָר	מזמן, לפני זמן רב
7	לְשֶׁעָבַר	לפנים, קודם לכן
8	חֲדָשׁוֹת לַבְּקָרִים	לעיתים קרובות, שוב ושוב
9	קַל וָחֹמֶר	על אחת כמה וכמה
10	מִבְּעוֹד מוֹעֵד	מראש, בזמן
11	בְּעִצּוּמוֹ שֶׁל	באמצע, בשיאו של
12	בִּמְשׂוּרָה	בצמצום, במידה מועטה
13	חַמְדָן	תאב ממון ורכוש
14	גַּנְדְּרָן	מתהדר בלבושו ובמראהו
15	קַנְטְרָן	מתגרה, מחפש ריב
16	עִלֵּג	מדבר בשיבושים, לא רהוט
17	נִלְעָג	מגוחך, עלוב, בזוי
18	זָחוּחַ	יהיר, מרוצה מעצמו
19	כִּילַי	קמצן, חס על כספו
20	אָמִיד	עשיר, מבוסס כלכלית
21	אֶבְיוֹן	עני מאוד, נזקק
22	כָּחוּשׁ	רזה מאוד
23	צָנוּם	רזה ודק
24	תִּמְהוֹנִי	מוזר, יוצא דופן
25	נוֹטֵר	שומר כעס בליבו
26	תַּכְסִיסָן	פועל בתחבולות
27	סוֹרֵר	מרדן, מפר משמעת
28	נִרְפֶּה	עצל, חסר מרץ
29	אָנִין	בעל טעם מעודן (אנין טעם)
30	אִסְטְנִיס	רגיש ומפונק
31	הֶדְיוֹט	אדם פשוט, חסר ידע
32	בּוּר	חסר השכלה; שדה לא מעובד (אדמת בור)
33	אֱוִיל	טיפש, חסר דעת
34	חָמוּם	רתחן, כועס בקלות (חמום מוח)
35	שְׁלוּמִיאֵל	מגושם וחסר מזל
36	הִתְנַכֵּל	זמם לפגוע, הציק
37	הִתְעַמֵּר	התאכזר, ניצל לרעה
38	הִתְנַצֵּחַ	התווכח בלהט
39	הִתֵּל	לעג, שיטה בו
40	הוֹנָה	רימה, עשק בממון
41	עָשַׁק	גזל, ניצל חלש
42	נִשֵּׁל	גירש מרכושו או מאדמתו
43	נָחַל	קיבל לחלקו, ירש (נחל אכזבה)
44	הִסְתּוֹפֵף	שהה במחיצת, מצא מקלט
45	חָסָה	שהה תחת הגנה, מצא מקלט (חסה בצילו)
46	כָּמַהּ	השתוקק מאוד
47	עָרַג	נכסף, השתוקק
48	נָהָה	נמשך אחרי
49	תָּר	סייר, חיפש (תר אחר מנוחה)
50	חִזֵּר	ניסה לזכות באהדה, ביקש קרבה (חיזר אחריה)
51	אָרַב	המתין במסתור
52	הִתְחַקָּה	עקב וחקר (התחקה אחר שורשיו)
53	הִפְטִיר	אמר כבדרך אגב
54	נָם	ישן, נרדם (נם את שנתו)
55	שִׁנֵּן	חזר ולמד היטב
56	נָבַט	החל לצמוח, בצבץ מהאדמה (הזרע נבט)
57	נָעַץ	תקע; הביט בחדות (נעץ מבט)
58	לָטַשׁ	שייף וחידד; הביט בחשק (לטש עיניים)
59	בָּהָה	הביט בריקנות
60	גָּהַר	כפף גופו מעל, גחן (גהר מעל הבאר)
