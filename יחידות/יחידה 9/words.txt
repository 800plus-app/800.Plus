# יחידה 9 — פסיכומטרי קמפוס, המילון הרשמי. מילה אחת = שורה אחת (כל הפירושים מאוחדים).

אֶבֶן יְסוֹד - דבר מרכזי במשהו, מרכז הכובד
אַבְּסוּרְדִי - חסר הגיון, שטותי, דבר שאינו מתקבל על הדעת
אַדְּרַבָּה - ההיפך הוא הנכון; למה לא? יופי!
אוֹב - קסם, כישוף
אוֹטוֹרִיטָה - סמכות הנובעת מידע בתחום מסוים
אוֹרַח - מנהג, אופן התנהגות, דרך חיים
אֵין תּוֹכוֹ כְּבָרוֹ - צבוע, שקרן
אִיֵּשׁ - הציב אנשים למילוי תפקיד מסוים
אִישׁ אֶשְׁכּוֹלוֹת - אדם הבקי בתחומים רבים ושונים
אֶנִיגְמָה - תעלומה, חידה
אַפְּרַקְדָן - שכוב על הגב
אֶשְׁתָּקַד - בשנה שעברה
אֶתְנַחְתָּא - הפסקה, פסק זמן
בְּחוּבּוֹ - בו, בתוכו
בִּיזָארִי - מוזר, לא נורמאלי
בִּכָּה - ספד, קונן והתאבל על מת
בַּלְדָּר - שליח, דוור, רץ
בָּלוּי - שחוק, מרופט, ישן וקרוע
בְּנַפְשׁוֹ הַדָּבָר - עבורו מדובר בדבר חיוני וגורלי בבחינת ייהרג ובל יעבור
בִּקֵּשׁ עַל נַפְשׁוֹ - התחנן על חייו
גְּדִיל - קצוות חוטים היוצאים משולי הבגד
גִּדֵם / גִּדֵעַ - קטוע יד; גזע העץ שנשאר לאחר כריתה, חלק מאיבר שנשאר לאחר קטיעה; כרת, חטב, קטע
גְּלִימָה - בגד עליון ארוך
גָּמַר עָלָיו אֶת הַהַלֵּל - שיבח אותו
דִּבְרֵי חֲלָקוֹת - דברי חנופה, לשון חלקות
דָּחָה אוֹתוֹ בְּקַש - התחמק מלתת תשובה בתירוץ קלוש
דֹּחַק - מצב של מחסור ומצוקה; צפיפות
דָּמַם - השתתק, לא הוציא הגה; הפסיק לזוז, נרגע, שקט
הֶאֱפִיל - עלה עליו ביכולתו; החשיך, הקדיר
הֲגַם שֶׁ- - למרות שֶׁ-, אף על פי, אם כי
הֵחִישׁ - האיץ, זירז
הֶחֱשָה - לא השמיע קול, שתק, דמם לא התערב
הִכְלִים - בייש, הביך, הלבין את פניו, עלב
הִכְתִּיר - הסמיך מלך חדש
הִמְרָה אֶת פִּיו - נהג בניגוד להוראתו
הֵסִיר אֶת הַלּוֹט - חשף, גילה, פרסם (לוט - כיסוי)
הִסְתּוֹפֵף - שהה באופן ממושך במקום מסוים או בחברת אנשים מסוימים
הִסְתּוֹפֵף בְּצִלּוֹ - היה לתלמידו (הסתופף - שהה במחיצת אנשים)
הֶעֱרָה - גילה, חשף; שתה, לגם
הִצְמִית - השמיד, חיסל, החריב
הִקְצָה - ייעד משהו למטרה מסוימת, הקציב, הפריש
הִרְעִים - הרגיז, הביא לידי כעס; השמיע קול רעם, הרעשה
הִתְאוֹנֵן - התלונן
הִתְגַּפֵּף - התחבק מתוך אהבה, התרפק, התלטף
התקין - הכין, סידר, יצר; קבע תקנה, חוקק
ווַרְיַאצְיָה - נוסח, גירסה
וִירְטוּאוֹז - המצטיין בשלמות במקצועו, בעל רמת ביצוע גבוה במיוחד
זִכָּיוֹן - זכות מיוחדת לדבר מה
זַלְזַל - ענף
חָפַן - תפס דבר מה בכף ידו; מידת נפח המסמלת את הכמות שניתן להחזיק בכף יד קמוצה
חָרַץ - גזר, פסק, קבע; שרט, עשה חריץ במשטח
טְפִיפָה - הליכה בצעדים קלים תוך התנדנדות, צעידה מרחפת
טַרְזָנִי - גנדרני, מתלבש בהידור
יִעוּד - הקצבת דבר או אדם למטרה מסוימת, הקדשה, יחוד; שליחות, כוונה, תכלית
יָצָא חוֹצֵץ - התנגד בתוקף
יָצָא לָשׂוּחַ - יצא לטייל
כִּילַי - קמצן
כָּמוּס - מוסתר, חסוי, מוחבא
כְּרוּב - מלאך
לָאַט - לחש, דיבר לאט ובלחש
לוֹט - כיסוי, מעטה; מכוסה, עטוף; מצורף, נספח
לוּט בָּעֲרָפֶל - שאינו ברור, מעורפל
לָחַך - נגע קלות (כשנאמר על גלים, אש וכו'); כשנאמר על חיה: אכל עשב; שרף לחלוטין, כילה
לְלֹא כָּחָל וְסָרָק - כמו שהוא מבלי להוסיף או לגרוע, ללא קישוטים ותוספות
לְמַפְרֵע - לאחר מעשה, על מה שהיה
לְמִשְעִי - באופן נקי ביותר, באופן חלק
לִסְטִים - שודד, גנב
לִפְנַי וְלִפְנִים - באופן יסודי ומעמיק
מִגְדָּנוֹת - דברי מתיקה (אפשר גם: מגדנים)
מָגוֹף / מְגוּפָה - ברז, פקק, שסתום ויסות
מוּבְהָק - ניכר בברור, בולט; אופייני, טיפוסי, ייצוגי
מוּפְשָט - ערטילאי, נטול גשמיות (להיפך: מוחשי)
מִינוּת - כפירה בדת, כפירה בעיקר
מְלוּבָּן - מובהר, מוסבר, מפורש
מִנְשָר - הצהרה, כרוז
מַעֲקָשׁ - מכשול, סיבוך; מפותל, שאינו מישורי
מְשׁוֹבֵב נֶפֶשׁ - מהנה ביותר, נעים
מִשְׁנֶה - כפול ומכופל, רב מאוד (משנה תוקף, משנה חשיבות)
מִתְוֶוה - שרטוט או רישום בקווים כלליים, סקיצה
נֵאוֹת - הסכים; מתאים, ראוי, הולם
נֹגַהּ - אור, זוהר
נָד / הִתְנוֹדֵד - נע, מתנועע
נִדְבָּך - דרגה בהתפתחותו של דבר מה, שלב; שכבה
נִטַּשׁ - התחולל, התפשט
נִכְמַר לִבּוֹ - ריחם, חס
נָסַךְ - יצק נוזלים לצורך פולחן דתי; התיך; השרה, יצר אוירה מסויימת
נִקְהַל - נקבץ, נאסף
נִקְרָה - סדק רחב בסלע או בהר, נַקִיק; הזדמן, הגיע למקום או נפגש עם מישהו באקראי, באופן לא מתוכנן (נקרה בדרכו)
סְבָרָה - דעה רווחת, השערה
סוֹלְלָה - גבעה מלאכותית מוארכת; יחידת תותחים
סָלַד - הרגיש דחייה כלפי משהו, נגעל
סָעַד - סייע, טיפל
סָעַד (אֶת) לִבּוֹ - אכל, השביע רעבונו
סָפַק אֶת כָּפָּיו - מחא כף מתוך צער או אכזבה
עַבְדְקָן - בעל זקן עבה ומגודל
עֲבוֹט - חפץ בעל ערך הניתן למִשכון בתמורה להלוואה, ערבון
עִוְעִים - תעתועים, שיגעון (רעיונות עִוְעִים - רעיונות מטורפים)
עוֹפֶר - אייל או צבי צעיר; כשנאמר על אדם: עלם חמד, צעיר
עַמוּד הַתָּוֶוך - דבר מרכזי במשהו, מרכז הכובד
עֵרָה - מזג, שפך נוזל; הרס, החריב; חשף, גילה
פֹּארָה - ענף
פּוֹחֵז - חסר אחריות, פזיז, נמהר
פֶּלֶך - כְּישוֹר, מוט המשמש לטווית חוטים; אזור, מחוז
פִּנְכָּה - קערה, צלוחית, פלטה (מְלַחֵךְ פִּנְכָּה - מתחנף)
פַּס - עבר מן העולם, תם
פָּקַד - בא, ביקר מישהו; סָפַר, מנה; ציווה, הורה
צָבָא - התאסף, התקהל
צוֹעֵר - חניך בית ספר לקצינים; עוזר, סגן
צוּר - סלע גדול
צוּר מַחְצַבְתּוֹ - מקורו, מולדתו, המקום שממנו בא ואליו הוא משתייך
צִיַה - מדבר, שממון
צָנוּף - מגולגל, מכורבל, מכווץ; כרוך סביב הראש
צַר - אויב, שונא; חֲבָל, מצער; לא רחב, מצומצם
קַב - מידת נפח קדומה
קְדֵרָה - כלי לבישול, סיר
קוֹלֶגָה - חבר למקצוע, עמית
קָטוּם - שנכרת ונקצץ ראשו
קָם - אויב; נבנה, יוסד; התעורר, פקח עינים
קַנְקַן זָפוּת - בעל זיכרון טוב (כמו קנקן שחוריו סתומים בזפת ואינו דולף)
קְסַרְקְטִין / קְסַרְקָט - מחנה צבאי, מגורי חיילים
קָרוּץ - עשוי מ-
רָבַץ לְפִתְחוֹ - נמצא בסביבתו הקרובה; ארב לו
רֵיאָלִי - ישים, אפשרי; מציאותי
רַעֲמָה - המון שיער; שיער ארוך הגדל על העורף של בעל חיים
רָצַע - עשה חור (מַרְצֵע - מכשיר לניקוב חורים)
שָׁוְא - שקר וכזב (לרוב מופיע בצירופי סמיכות כגון "תלונת שווא")
שָׁוֶוה לְכֹל נֶפֶשׁ - כל אחד יכול להרשות לעצמו, זול; מתאים לכל אחד, פשוט
שֶׁנֶץ - רצועה, שרוך
שֶׁצֶף - זרם חזק, שטף; כעס, זעם
שְׁרִירוּת - אופן התנהגותו של מי שפועל לפי גחמותיו גם אם הן פוגעות באחרים
שְׁרִירוּת לֵב - עקשנות, התנהגות חסרת הגיון והתחשבות
שְׁרִירוּתִי - בנוי על דעה בלתי מבוססת או רצון עיקש שאין לו הצדקה אמיתית
שָׂרָף - כינוי לנחש ארסי; מלאך
תּוֹאֲנָה - סיבה, תירוץ
תּוֹכָחָה / תּוֹכֵחָה - דברי מוסר, עונש
תּוֹלָדָה - תוצאה, מסקנה, דבר שהסתעף מדבר עיקרי
תְּכִיפוּת - תדירות, נפיצות
תָּכַף - התרחש לעתים קרובות, בתדירות גבוהה
תָּר - ריגל, בלש אחרי, עקב; חיפש, ביקש; סייר, טייל
תְּשׁוּרָה - מתנה, מנחה, שי
